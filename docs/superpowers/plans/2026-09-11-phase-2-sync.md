> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The Items widgets work. One scheduler loop feeds a per-environment SQLite item cache from per-connection adapters (GitHub and the mock source); widgets read the cache and never fetch; `item_action` is the single outbound path and logs an event; OAuth consent runs on a loopback listener with PKCE.

`src/core/repos/remote.ts`:

```ts
/**
 * Turning every spelling of a git remote into one stable key. Port of
 * `repos/remote.rs`. This is the join between "a repo on GitHub" and "a folder
 * on disk"; if it is wrong, every repo silently reports as unmapped.
 */

/** `host/owner/repo`, lowercased. The host is part of the key on purpose. */
export function normalize(rawUrl: string): string | null {
  const url = rawUrl.trim();
  if (url === "") return null;

  // Drop the scheme, then any userinfo (`git@`, `user:token@`).
  const schemeAt = url.indexOf("://");
  const withoutScheme = schemeAt >= 0 ? url.slice(schemeAt + 3) : url;
  const at = withoutScheme.lastIndexOf("@");
  const withoutUser = at >= 0 ? withoutScheme.slice(at + 1) : withoutScheme;

  // Split host from path at the first ':' or '/' — except a bracketed IPv6
  // host has to be taken as one unit first.
  let host: string;
  let remainder: string;
  if (withoutUser.startsWith("[")) {
    const close = withoutUser.indexOf("]");
    if (close < 0) return null;
    host = withoutUser.slice(0, close + 1);
    remainder = withoutUser.slice(close + 1);
  } else {
    const index = withoutUser.search(/[:/]/);
    if (index < 0) return null;
    host = withoutUser.slice(0, index);
    remainder = withoutUser.slice(index);
  }
  if (remainder === "") return null;
  const separator = remainder[0];
  remainder = remainder.slice(1);

  // After a ':' the next segment is either a port (`host:22/owner/repo`) or
  // the start of an scp-style path (`host:owner/repo`).
  let path = remainder;
  if (separator === ":") {
    const slash = remainder.indexOf("/");
    if (slash > 0) {
      const maybePort = remainder.slice(0, slash);
      if (/^\d+$/.test(maybePort)) path = remainder.slice(slash + 1);
    }
  }

  path = path.replace(/\/+$/, "");
  // Lowercase before stripping `.git`, not after.
  path = path.toLowerCase();
  if (path.endsWith(".git")) path = path.slice(0, -4);
  if (host === "" || path === "" || !path.includes("/")) return null;
  return `${host.toLowerCase()}/${path}`;
}

/** `owner/repo`, ignoring the host — the weaker second tier. A match made
 *  this way is flagged in the UI rather than presented as certain. */
export function fallbackKey(rawUrl: string): string | null {
  let cleaned = rawUrl.trim().replace(/\/+$/, "").toLowerCase();
  if (cleaned.endsWith(".git")) cleaned = cleaned.slice(0, -4);
  const segments = cleaned.split(/[/:]/).filter((s) => s !== "").reverse();
  const repo = segments[0];
  const owner = segments[1];
  if (repo === undefined || owner === undefined) return null;
  if (owner.includes(".") && !owner.includes("/") && segments[2] === undefined) {
    // `github.com/ddd` — the "owner" is really the host; not enough to go on.
    return null;
  }
  return `${owner}/${repo}`;
}
```

`src/core/sync/signal.ts`:

```ts
/**
 * Signal, not the firehose. Port of `sync/signal.rs`.
 *
 * A scorer rather than a boolean: mail is genuinely ambiguous in the middle,
 * and a score with reasons attached is something a person can disagree with.
 * The weighting rests on one distinction: structural evidence (who sent it,
 * which headers it carries) outranks lexical evidence (words in the subject).
 */

export interface MailSignals {
  from: string;
  subject: string;
  /** Lowercased header names present on the message. */
  headers: string[];
  /** True when the recipient is on `To:` rather than only Cc/Bcc. */
  addressedDirectly: boolean;
  /** True when this thread already contains a message from the user. */
  repliedBefore: boolean;
}

export function mailSignals(partial: Partial<MailSignals> & Pick<MailSignals, "from" | "subject">): MailSignals {
  return { headers: [], addressedDirectly: false, repliedBefore: false, ...partial };
}

export interface Verdict {
  signal: boolean;
  score: number;
  /** Machine-readable, e.g. `noise:list-unsubscribe`. */
  reasons: string[];
}

/** Above this, a human probably wrote it and probably wants an answer. */
const THRESHOLD = 0;
/** Facts about the message: sender identity, list headers. */
const STRUCTURAL = 4;
/** Guesses about its meaning: words in the subject. */
const LEXICAL = 2;
/** Being written to rather than cc'd. Must outweigh one lexical hit. */
const ADDRESSED = 3;

const ROBOT_LOCALS = [
  "noreply", "no-reply", "no_reply", "donotreply", "do-not-reply", "notifications",
  "notification", "mailer-daemon", "bounce", "bounces", "newsletter", "news", "marketing",
  "billing", "receipts", "invoice", "alerts", "alert", "automated", "auto-confirm", "jenkins",
  "build", "ci", "postmaster", "support-noreply",
];

const ROBOT_HEADERS = [
  "list-unsubscribe", "list-id", "auto-submitted", "x-auto-response-suppress", "precedence",
  "x-campaign-id", "x-mailer-daemon", "feedback-id",
];

/** Subject shapes that *suggest* automation. Weak evidence by design. */
const ROBOT_SUBJECTS = [
  "build failed", "build passed", "build succeeded", "pipeline failed", "deployment succeeded",
  "your receipt", "invoice", "order confirmation", "password reset", "verify your email",
  "unsubscribe", "weekly digest", "daily digest", "newsletter", "security alert",
  "terms of service", "privacy policy",
];

export function classify(mail: MailSignals): Verdict {
  let score = 0;
  const reasons: string[] = [];

  const from = mail.from.toLowerCase();
  // The Rust: last segment after '<', then before '@', trimmed of anything
  // that is not [a-z0-9_-].
  const afterAngle = from.slice(from.lastIndexOf("<") + 1);
  const local = (afterAngle.split("@")[0] ?? "").replace(/^[^a-z0-9_-]+|[^a-z0-9_-]+$/g, "");

  // Structural: who actually sent it.
  if (ROBOT_LOCALS.some((robot) => local === robot || local.startsWith(`${robot}-`))) {
    score -= STRUCTURAL;
    reasons.push(`noise:sender-${local}`);
  }

  for (const raw of mail.headers) {
    const header = raw.toLowerCase();
    if (ROBOT_HEADERS.includes(header)) {
      // List-Unsubscribe is the single most reliable tell.
      score -= header === "list-unsubscribe" ? STRUCTURAL : Math.trunc(STRUCTURAL / 2);
      reasons.push(`noise:${header}`);
    }
  }

  const subject = mail.subject.toLowerCase();
  // Lexical: weaker than a direct address on purpose.
  const hit = ROBOT_SUBJECTS.find((pattern) => subject.includes(pattern));
  if (hit !== undefined) {
    score -= LEXICAL;
    reasons.push(`noise:subject-${hit.replaceAll(" ", "-")}`);
  }

  // Bracketed prefixes are how ticket systems announce themselves: [JIRA].
  if (subject.startsWith("[") && subject.includes("]")) {
    score -= 1;
    reasons.push("noise:bracket-prefix");
  }

  if (mail.addressedDirectly) {
    score += ADDRESSED;
    reasons.push("signal:addressed-to-me");
  }
  if (mail.repliedBefore) {
    // You have already engaged with this thread; it is yours now.
    score += 5;
    reasons.push("signal:in-my-thread");
  }

  return { signal: score > THRESHOLD, score, reasons };
}

/** The item kind a verdict implies; "real mail only" is `WHERE kind = 'mail'`. */
export function kindFor(verdict: Verdict): "mail" | "mail_bulk" {
  return verdict.signal ? "mail" : "mail_bulk";
}
```

```ts
 * The adapter contract. Port of `sync/adapter.rs`.
 *
 * One interface, one normalized `Item`, one place where a new integration
 * plugs in. Adapters convert *into* the shared shape and nothing downstream
 * ever learns which service an item came from. Instantiated per connection,
 * not per provider — work GitHub and personal GitHub are two instances.
 *
 * What the Rust took from Tauri's managed state (the HTTP client) and the OS
 * (keychain, browser) arrives here as `SyncDeps`, so core stays free of
 * Electron and every adapter is testable with a fake `fetch`.
 */
import { DddError } from "../error";
import type { Emitter } from "../emitter";
import type { EnvRoot } from "../paths";
import type { Connection } from "../settings";
import type { Item } from "../models";

export interface SecretStore {
  get(env: EnvRoot, key: string): string | null;
  set(env: EnvRoot, key: string, value: string): void;
}

/** What the sync engine needs from the host. */
export interface SyncDeps {
  fetch: typeof globalThis.fetch;
  secrets: SecretStore;
  openUrl(url: string): Promise<void>;
  emit: Emitter;
}

export interface FetchContext {
  environmentId: string;
  connection: Connection;
  /** Read from the keychain at fetch time and never handed to anything else. */
  token: string;
  fetch: typeof globalThis.fetch;
}

/** Connection-specific settings the user typed in, kept as loose strings so a
 *  new adapter never needs a change to the settings schema. */
export function configOf(cx: FetchContext, key: string): string | null {
  const v = cx.connection.config[key];
  return v === undefined ? null : v;
}

export function requireConfig(cx: FetchContext, key: string): string {
  const v = configOf(cx, key);
  if (v === null || v === "") throw DddError.config(`${key} is not set for ${cx.connection.label}`);
  return v;
}

/** What the user is asking the adapter to do to one item. Every outbound
 *  variant is reachable only from an explicit click or keypress. */

  action: string;
  item: Item;
  /** Comment body, transition name — whatever the action needs. */
  input: string | null;
}

  /** Matches `Item.source`, e.g. `github`. */
  readonly source: string;
  /** Pull everything this connection contributes, already normalized. */
  fetch(cx: FetchContext): Promise<Item[]>;
  /** Perform a user-approved action. Default: politely refuse. */
  perform(cx: FetchContext, request: ActionRequest): Promise<string>;
}

export function refuse(source: string, action: string): never {

}
```

```ts
/**
 * The normalized item cache, one database per environment. Port of
 * `sync/cache.rs`. Widgets read from here and only from here; the scheduler is
 * the only writer. Same DDL as the Rust so both apps share the file.
 */
import { DatabaseSync } from "node:sqlite";
import { DddError } from "../error";
import type { EnvRoot } from "../paths";
import type { Item, ItemAction, ItemFilter } from "../models";

const DDL = `
  PRAGMA journal_mode=WAL;
  CREATE TABLE IF NOT EXISTS items (
     id            TEXT PRIMARY KEY,
     connection_id TEXT NOT NULL,
     source        TEXT NOT NULL,
     kind          TEXT NOT NULL,
     project       TEXT,
     title         TEXT NOT NULL,
     body          TEXT,
     url           TEXT,
     refs          TEXT NOT NULL DEFAULT '[]',
     actions       TEXT NOT NULL DEFAULT '[]',
     updated_at    TEXT NOT NULL,
     unread        INTEGER NOT NULL DEFAULT 0,
     synced_at     TEXT NOT NULL
  );
  CREATE INDEX IF NOT EXISTS items_conn    ON items(connection_id);
  CREATE INDEX IF NOT EXISTS items_source  ON items(source, kind);
  CREATE INDEX IF NOT EXISTS items_updated ON items(updated_at DESC);`;

function withDb<T>(env: EnvRoot, body: (db: DatabaseSync) => T): T {
  let db: DatabaseSync;
  try {
    // rusqlite sets a 5 s busy timeout on every open; node:sqlite defaults to 0.
    db = new DatabaseSync(env.cacheDb(), { timeout: 5000 });
  } catch (e) {
    throw DddError.sqlite(e);
  }
  try {
    db.exec(DDL);
    return body(db);
  } catch (e) {
    throw e instanceof DddError ? e : DddError.sqlite(e);
  } finally {
    db.close();
  }
}

/** Replace everything this connection contributed with a fresh snapshot, in
 *  one transaction so a widget can never observe the gap. A snapshot rather
 *  than a merge, because "closed yesterday" has to be able to disappear. */
export function replaceForConnection(env: EnvRoot, connectionId: string, items: Item[]): void {
  withDb(env, (db) => {
    db.exec("BEGIN");
    try {
      db.prepare("DELETE FROM items WHERE connection_id = ?").run(connectionId);
      const syncedAt = new Date().toISOString();
      // OR REPLACE: a duplicate id from a misbehaving adapter costs that one
      // row, not the whole transaction.
      const insert = db.prepare(
        `INSERT OR REPLACE INTO items
           (id, connection_id, source, kind, project, title, body, url, refs, actions, updated_at, unread, synced_at)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      );
      for (const item of items) {
        insert.run(
          item.id, item.connectionId, item.source, item.kind, item.project ?? null, item.title,
          item.body ?? null, item.url ?? null, JSON.stringify(item.refs), JSON.stringify(item.actions),
          item.updatedAt, item.unread ? 1 : 0, syncedAt,
        );
      }
      db.exec("COMMIT");
    } catch (e) {
      db.exec("ROLLBACK");
      throw e;
    }
  });
}

const COLUMNS = "id, connection_id, source, kind, project, title, body, url, refs, actions, updated_at, unread";

interface Row {
  id: string; connection_id: string; source: string; kind: string; project: string | null;
  title: string; body: string | null; url: string | null; refs: string; actions: string;
  updated_at: string; unread: number;
}

function rowToItem(r: Row, environmentId: string): Item {
  return {
    id: r.id,
    // Stamped from the root that opened the database: a row physically cannot
    // have come from another environment.
    environmentId,
    connectionId: r.connection_id,
    source: r.source,
    kind: r.kind,
    project: r.project,
    title: r.title,
    body: r.body,
    url: r.url,
    refs: parseJson<string[]>(r.refs, []),
    actions: parseJson<ItemAction[]>(r.actions, []),
    updatedAt: r.updated_at,
    unread: r.unread !== 0,
  };
}

function parseJson<T>(text: string, fallback: T): T {
  try {
    return JSON.parse(text) as T;
  } catch {
    return fallback;
  }
}

export function query(env: EnvRoot, filter: ItemFilter): Item[] {
  return withDb(env, (db) => {
    const like = filter.query ? `%${filter.query}%` : null;
    const excludeKinds = filter.excludeKinds ?? [];
    // Built into the SQL, not filtered afterward: exclusion has to happen
    // before LIMIT or a caller asking for 200 items with one kind excluded can
    // get back far fewer than 200.
    const excludeClause = excludeKinds.length === 0 ? "" : `AND kind NOT IN (${excludeKinds.map(() => "?").join(",")})`;
    const sql = `SELECT ${COLUMNS} FROM items
      WHERE (? IS NULL OR connection_id = ?)
        AND (? IS NULL OR source = ?)
        AND (? IS NULL OR kind = ?)
        AND (? IS NULL OR project = ?)
        AND (? IS NULL OR title LIKE ? OR IFNULL(body,'') LIKE ?)
        AND (? = 0 OR unread = 1)
        ${excludeClause}
      ORDER BY updated_at DESC
      LIMIT ?`;
    const c = filter.connectionId ?? null;
    const s = filter.source ?? null;
    const k = filter.kind ?? null;
    const p = filter.project ?? null;
    const unread = filter.unreadOnly ? 1 : 0;
    const params: (string | number | null)[] = [c, c, s, s, k, k, p, p, like, like, like, unread, ...excludeKinds, filter.limit ?? 200];
    const rows = db.prepare(sql).all(...params) as unknown as Row[];
    return rows.map((r) => rowToItem(r, env.id));
  });
}

/** Look one item up by id, for acting on it. */
export function find(env: EnvRoot, id: string): Item | null {
  return withDb(env, (db) => {
    const row = db.prepare(`SELECT ${COLUMNS} FROM items WHERE id = ?`).get(id) as unknown as Row | undefined;
    return row ? rowToItem(row, env.id) : null;
  });
}

export function clear(env: EnvRoot): void {
  withDb(env, (db) => {
    db.prepare("DELETE FROM items").run();
  });
}
```

```ts
/**
 * GitHub, the first real adapter. Port of `sync/github.rs`.
 *
 * Everything the board needs about pull requests and issues comes back in a
 * single GraphQL request, including the CI rollup. One more REST call fetches
 * notifications, and a paginated GraphQL call lists repositories. That budget
 * is the whole reason the scheduler is central.
 */
import { DddError } from "../error";
import type { Item, ItemAction } from "../models";
import { normalize } from "../repos/remote";
import { itemId, type ActionRequest, type FetchContext, type SourceAdapter } from "./adapter";

export const API = "https://api.github.com";
const UA = "ddd-dashboard";
const TIMEOUT_MS = 30_000;

export const QUERY = `
query {
  authored: search(query: "is:pr is:open author:@me", type: ISSUE, first: 30) { nodes { ...pr } }
  review:   search(query: "is:pr is:open review-requested:@me", type: ISSUE, first: 30) { nodes { ...pr } }
  issues:   search(query: "is:issue is:open assignee:@me", type: ISSUE, first: 30) { nodes { ...iss } }
}
fragment pr on PullRequest {
  number title url updatedAt isDraft reviewDecision
  repository { nameWithOwner }
  commits(last: 1) { nodes { commit { statusCheckRollup { state } } } }
}
fragment iss on Issue {
  number title url updatedAt
  repository { nameWithOwner }
}
`;

export const REPOS_QUERY = `
query($cursor: String) {
  viewer {
    repositories(first: 100, after: $cursor,
                 orderBy: {field: PUSHED_AT, direction: DESC},
                 affiliations: [OWNER, COLLABORATOR, ORGANIZATION_MEMBER]) {
      pageInfo { hasNextPage endCursor }
      nodes {
        nameWithOwner url sshUrl pushedAt isPrivate isArchived isFork
        primaryLanguage { name }
      }
    }
  }
}
`;

/** Pages of 100. Three is the stated cap from the spec. */
export const REPO_PAGES = 3;

type Json = Record<string, unknown>;
const obj = (v: unknown): Json | null => (typeof v === "object" && v !== null && !Array.isArray(v) ? (v as Json) : null);
const str = (v: unknown): string | null => (typeof v === "string" ? v : null);
const num = (v: unknown): number | null => (typeof v === "number" ? v : null);
const bool = (v: unknown, fallback: boolean): boolean => (typeof v === "boolean" ? v : fallback);
/** `a.b.c` lookup over untyped JSON. */
function at(v: unknown, path: string): unknown {
  let cur: unknown = v;
  for (const key of path.split(".")) {
    const o = obj(cur);
    if (!o) return undefined;
    cur = o[key];
  }
  return cur;
}

function headers(cx: FetchContext): Record<string, string> {
  return { authorization: `Bearer ${cx.token}`, "user-agent": UA, accept: "application/vnd.github+json" };
}

function net(e: unknown): DddError {
  return DddError.config(`github request failed: ${e instanceof Error ? e.message : String(e)}`);
}

async function send(cx: FetchContext, url: string, init: RequestInit): Promise<Response> {
  try {
    return await cx.fetch(url, { ...init, headers: { ...headers(cx), ...(init.headers as Record<string, string> | undefined) }, signal: AbortSignal.timeout(TIMEOUT_MS) });
  } catch (e) {
    throw net(e);
  }
}

/** Turn a non-2xx into a message a human can act on. GitHub puts the useful
 *  part in the body, and a bare status code helps nobody. */
async function check(response: Response): Promise<unknown> {
  const text = await response.text().catch(() => "");
  if (!response.ok) {
    let detail = text.slice(0, 200);
    try {
      const m = str(obj(JSON.parse(text))?.message);
      if (m !== null) detail = m;
    } catch {
      /* not JSON */
    }
    throw DddError.config(`github ${response.status} ${response.statusText}: ${detail}`.replace(/\s+:/, ":"));
  }
  try {
    return JSON.parse(text);
  } catch {
    return null;
  }
}

/** GraphQL with an optional pagination cursor bound to `$cursor`. GraphQL
 *  answers 200 with an `errors` array, so a happy status proves nothing. */
async function graphql(cx: FetchContext, query: string, cursor: string | null = null): Promise<unknown> {
  const response = await send(cx, `${API}/graphql`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ query, variables: { cursor } }) });
  const body = await check(response);
  const errors = at(body, "errors");
  if (Array.isArray(errors) && errors.length > 0) {
    const message = errors.map((e) => str(obj(e)?.message)).filter((m): m is string => m !== null).join("; ");
    throw DddError.config(`github graphql: ${message}`);
  }
  return at(body, "data") ?? null;
}

async function post(cx: FetchContext, url: string, body: unknown): Promise<unknown> {
  const response = await send(cx, url, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) });
  return check(response);
}

export function toItem(cx: FetchContext, node: unknown, kind: string): Item | null {
  const repo = str(at(node, "repository.nameWithOwner"));
  const number = num(at(node, "number"));
  const title = str(at(node, "title"));
  const updated = str(at(node, "updatedAt"));
  if (repo === null || number === null || title === null || updated === null) return null;
  const url = str(at(node, "url"));

  const ci = str(at(node, "commits.nodes.0.commit.statusCheckRollup.state"));
  const review = str(at(node, "reviewDecision"));
  const draft = bool(at(node, "isDraft"), false);

  // The status line is folded into `body` so every view gets it for free and
  // the Item model does not sprout source-specific columns.
  const status: string[] = [];
  if (draft) status.push("draft");
  if (ci !== null) status.push(`ci ${ci.toLowerCase()}`);
  if (review !== null) status.push(review.toLowerCase().replaceAll("_", " "));

  const actions: ItemAction[] = [
    { id: "comment", label: "Comment", outbound: true },
    { id: "open", label: "Open on GitHub", outbound: false },
  ];
  if (kind === "review_request") actions.unshift({ id: "approve", label: "Approve", outbound: true });

  return {
    id: itemId(cx.connection.id, `${repo}#${number}`),
    environmentId: cx.environmentId,
    connectionId: cx.connection.id,
    source: "github",
    kind,
    project: repo,
    title: `${repo}#${number} ${title}`,
    body: status.length > 0 ? status.join(" · ") : null,
    url,
    // The self-ref is what `perform` parses to know where to post back to.
    refs: [`github:${repo}#${number}`],
    actions,
    updatedAt: updated,
    unread: false,
  };
}

/** Notifications are the closest thing GitHub has to "someone is waiting on you". */
async function notifications(cx: FetchContext): Promise<Item[]> {
  const response = await send(cx, `${API}/notifications?participating=true&per_page=50`, { method: "GET" });
  const body = await check(response);
  if (!Array.isArray(body)) return [];
  const items: Item[] = [];
  for (const thread of body) {
    const id = str(at(thread, "id"));
    const title = str(at(thread, "subject.title"));
    const repo = str(at(thread, "repository.full_name"));
    const updated = str(at(thread, "updated_at"));
    if (id === null || title === null || repo === null || updated === null) continue;
    const reason = str(at(thread, "reason")) ?? "";
    const unread = bool(at(thread, "unread"), true);
    // The API URL is not clickable; the web URL is derivable from the number.
    const subjectUrl = str(at(thread, "subject.url"));
    const number = subjectUrl?.split("/").pop();
    items.push({
      id: itemId(cx.connection.id, `thread-${id}`),
      environmentId: cx.environmentId,
      connectionId: cx.connection.id,
      source: "github",
      kind: "comment",
      project: repo,
      title: `${repo}: ${title}`,
      body: reason.replaceAll("_", " "),
      url: number ? `https://github.com/${repo}/pull/${number}` : null,
      refs: [`github-thread:${id}`],
      actions: [
        { id: "mark_read", label: "Mark read", outbound: true },
        { id: "open", label: "Open on GitHub", outbound: false },
      ],
      updatedAt: updated,
      unread,
    });
  }
  return items;
}

/** Repositories as items, so they ride the same cache and scheduler. */
async function repositories(cx: FetchContext): Promise<Item[]> {
  const items: Item[] = [];
  let cursor: string | null = null;
  for (let page = 0; page < REPO_PAGES; page++) {
    const data = await graphql(cx, REPOS_QUERY, cursor);
    const repos = at(data, "viewer.repositories");
    const nodes = at(repos, "nodes");
    if (Array.isArray(nodes)) for (const node of nodes) { const item = toRepoItem(cx, node); if (item) items.push(item); }
    const hasNext = bool(at(repos, "pageInfo.hasNextPage"), false);
    const next = str(at(repos, "pageInfo.endCursor"));
    // A missing or unchanged cursor means the API cannot advance, even if it
    // still claims hasNextPage — looping again would refetch page one.
    if (next !== null && hasNext && next !== cursor) cursor = next;
    else break;
  }
  return items;
}

export function toRepoItem(cx: FetchContext, node: unknown): Item | null {
  const name = str(at(node, "nameWithOwner"));
  const url = str(at(node, "url"));
  if (name === null || url === null) return null;
  const sshUrl = str(at(node, "sshUrl")) ?? "";
  const pushed = str(at(node, "pushedAt")) ?? "";
  const isPrivate = bool(at(node, "isPrivate"), false);
  const archived = bool(at(node, "isArchived"), false);
  const fork = bool(at(node, "isFork"), false);
  const language = str(at(node, "primaryLanguage.name"));

  const facts = [isPrivate ? "private" : "public"];
  if (language !== null) facts.push(language.toLowerCase());
  if (fork) facts.push("fork");
  if (archived) facts.push("archived");

  // Both raw forms are kept so a local clone matches whichever the user used.
  const refs = [`repo:${name}`];
  const normalized = normalize(url);
  if (normalized !== null) refs.push(`remote:${normalized}`);
  refs.push(`https:${url}`);
  if (sshUrl !== "") refs.push(`ssh:${sshUrl}`);

  return {
    id: itemId(cx.connection.id, `repo-${name}`),
    environmentId: cx.environmentId,
    connectionId: cx.connection.id,
    source: "github",
    kind: "repo",
    project: name,
    title: name,
    body: facts.join(" · "),
    url,
    refs,
    actions: [{ id: "open", label: "Open on GitHub", outbound: false }],
    updatedAt: pushed,
    unread: false,
  };
}

/** Pull `owner/repo` and the number back out of the self-ref. */
export function selfRef(item: Item): [string, number] {
  for (const r of item.refs) {
    if (!r.startsWith("github:")) continue;
    const [repo, n] = r.slice("github:".length).split("#");
    const number = Number(n);
    if (repo && Number.isInteger(number)) return [repo, number];
  }
  throw DddError.config(`${item.title} has no GitHub reference`);
}

export class GitHubAdapter implements SourceAdapter {
  readonly source = "github";

  async fetch(cx: FetchContext): Promise<Item[]> {
    const items: Item[] = [];
    const data = await graphql(cx, QUERY);
    for (const [list, kind] of [["authored", "pull_request"], ["review", "review_request"], ["issues", "issue"]] as const) {
      const nodes = at(data, `${list}.nodes`);
      if (Array.isArray(nodes)) for (const node of nodes) { const item = toItem(cx, node, kind); if (item) items.push(item); }
    }
    items.push(...(await notifications(cx).catch(() => [] as Item[])));
    // A repo listing is not urgent; a failure here must not cost you your PRs.
    items.push(...(await repositories(cx).catch(() => [] as Item[])));
    // Duplicates are not guaranteed adjacent: a set over the whole result is
    // what actually guarantees first-writer-wins.
    const seen = new Set<string>();
    return items.filter((item) => (seen.has(item.id) ? false : (seen.add(item.id), true)));
  }

  async perform(cx: FetchContext, request: ActionRequest): Promise<string> {
    switch (request.action) {
      case "comment": {
        const [repo, number] = selfRef(request.item);
        if (request.input === null) throw DddError.config("a comment needs a body");
        await post(cx, `${API}/repos/${repo}/issues/${number}/comments`, { body: request.input });
        return `Commented on ${repo}#${number}`;
      }
      case "approve": {
        const [repo, number] = selfRef(request.item);
        await post(cx, `${API}/repos/${repo}/pulls/${number}/reviews`, { event: "APPROVE", body: request.input ?? "" });
        return `Approved ${repo}#${number}`;
      }
      case "mark_read": {
        const thread = request.item.refs.map((r) => (r.startsWith("github-thread:") ? r.slice("github-thread:".length) : null)).find((t) => t !== null);
        if (!thread) throw DddError.config("not a notification");
        const response = await send(cx, `${API}/notifications/threads/${thread}`, { method: "PATCH" });
        await check(response);
        return "Marked read";
      }
      default:
        throw DddError.config(`github cannot ${request.action}`);
    }
  }
}
```

```ts
/**
 * OAuth for a desktop app that has no server. Port of `sync/oauth.rs`.
 *
 * Loopback redirect + PKCE (RFC 8252). No client secret is trusted: PKCE is
 * what proves the exchange came from the process that started it. Tokens land
 * in the keychain, namespaced per environment; the access token goes in the
 * connection's main slot so adapters read it exactly as they read a pasted PAT.
 */
import { createHash, randomBytes } from "node:crypto";
import { createServer, type Server } from "node:http";
import type { AddressInfo } from "node:net";
import { DddError } from "../error";
import type { EnvRoot } from "../paths";
import { settingsLoad, settingsSave, type Connection } from "../settings";
import type { SecretStore, SyncDeps } from "./adapter";

/** How long we wait for the person to finish in the browser. */
export const CONSENT_TIMEOUT_MS = 180_000;

export interface OauthProvider {
  id: string;
  authUrl: string;
  tokenUrl: string;
  scopes: string;
  /** Providers that only hand out a refresh token when explicitly asked. */
  extraAuthParams: [string, string][];
}

export function providerFor(provider: string): OauthProvider | null {
  switch (provider) {
    case "mail.gmail":
    case "calendar.google":
      return {
        id: "google",
        authUrl: "https://accounts.google.com/o/oauth2/v2/auth",
        tokenUrl: "https://oauth2.googleapis.com/token",
        scopes: "https://www.googleapis.com/auth/gmail.modify https://www.googleapis.com/auth/calendar.readonly",
        // Google returns a refresh token only on the first consent unless you insist.
        extraAuthParams: [["access_type", "offline"], ["prompt", "consent"]],
      };
    case "slack":
      return {
        id: "slack",
        authUrl: "https://slack.com/oauth/v2/authorize",
        tokenUrl: "https://slack.com/api/oauth.v2.access",
        scopes: "channels:history,groups:history,im:history,mpim:history,users:read,chat:write,search:read",
        extraAuthParams: [],
      };
    default:
      return null;
  }
}

export function base64url(bytes: Uint8Array): string {
  return Buffer.from(bytes).toString("base64url");
}

const randomToken = () => base64url(randomBytes(32));

export function pkceChallenge(verifier: string): string {
  return base64url(createHash("sha256").update(verifier).digest());
}

/** `code=…&state=…` out of a request target; empty when there is no query. */
export function parseQuery(target: string): Map<string, string> {
  const out = new Map<string, string>();
  const q = target.indexOf("?");
  if (q < 0) return out;
  for (const pair of target.slice(q + 1).split("&")) {
    const eq = pair.indexOf("=");
    if (eq < 0) continue;
    try {
      out.set(pair.slice(0, eq), decodeURIComponent(pair.slice(eq + 1)));
    } catch {
      /* a value that does not decode is dropped, as the Rust's `.ok()?` drops it */
    }
  }
  return out;
}

/** Wait for the browser to come back with `?code=…`, verifying `state`. Any
 *  request without a query (a favicon) is answered 404 and ignored. */
export function awaitRedirect(server: Server, expectedState: string, timeoutMs = CONSENT_TIMEOUT_MS): Promise<string> {
  return new Promise((resolve, reject) => {
    const finish = (fn: () => void) => {
      clearTimeout(timer);
      server.close();
      fn();
    };
    const timer = setTimeout(() => finish(() => reject(DddError.config("timed out waiting for consent"))), timeoutMs);
    server.on("request", (req, res) => {
      const params = parseQuery(req.url ?? "/");
      if (params.size === 0) {
        res.writeHead(404, { "content-length": "0" }).end();
        return;
      }
      const stateOk = params.get("state") === expectedState;
      const body = !stateOk
        ? "Something went wrong: the state did not match. Nothing was connected."
        : params.has("code")
          ? "Connected. You can close this tab and go back to DDD."
          : "Consent was denied. Nothing was connected.";
      res.writeHead(200, { "content-type": "text/html; charset=utf-8", "content-length": String(Buffer.byteLength(body)) }).end(body);
      if (!stateOk) {
        // The exact shape of a CSRF attempt on the loopback listener: fail closed.
        finish(() => reject(DddError.config("OAuth state mismatch — nothing was connected")));
        return;
      }
      const code = params.get("code");
      finish(() => (code ? resolve(code) : reject(DddError.config("consent was denied"))));
    });
  });
}

async function postForm(fetch: typeof globalThis.fetch, url: string, form: [string, string][]): Promise<Record<string, unknown>> {
  const body = form.map(([k, v]) => `${encodeURIComponent(k)}=${encodeURIComponent(v)}`).join("&");
  let response: Response;
  try {
    response = await fetch(url, { method: "POST", headers: { "content-type": "application/x-www-form-urlencoded" }, body, signal: AbortSignal.timeout(30_000) });
  } catch (e) {
    throw DddError.config(`token request failed: ${e instanceof Error ? e.message : String(e)}`);
  }
  const text = await response.text().catch(() => "");
  let parsed: Record<string, unknown> = {};
  try {
    const j = JSON.parse(text);
    if (typeof j === "object" && j !== null) parsed = j as Record<string, unknown>;
  } catch {
    /* not JSON */
  }
  if (!response.ok) {
    const detail = (typeof parsed.error_description === "string" && parsed.error_description) || (typeof parsed.error === "string" && parsed.error) || text;
    throw DddError.config(`oauth ${response.status}: ${detail}`);
  }
  // Slack answers 200 with `ok: false`, so status alone proves nothing.
  if (parsed.ok === false) {
    throw DddError.config(`oauth: ${typeof parsed.error === "string" ? parsed.error : "rejected"}`);
  }
  return parsed;
}

function storeTokens(env: EnvRoot, connection: Connection, tokens: Record<string, unknown>, secrets: SecretStore): void {
  // Slack nests the user token; Google returns it flat.
  const nested = typeof tokens.authed_user === "object" && tokens.authed_user !== null ? (tokens.authed_user as Record<string, unknown>).access_token : undefined;
  const access = typeof tokens.access_token === "string" ? tokens.access_token : typeof nested === "string" ? nested : null;
  if (access === null) throw DddError.config("no access token in the response");
  secrets.set(env, connection.id, access);
  if (typeof tokens.refresh_token === "string") secrets.set(env, `${connection.id}:refresh`, tokens.refresh_token);

  const settings = settingsLoad(env);
  const stored = settings.connections.find((c) => c.id === connection.id);
  if (stored) {
    stored.status = "connected";
    stored.error = null;
    if (typeof tokens.expires_in === "number") {
      // Expire a minute early: a token that dies mid-request reads as "broken".
      stored.config.expiresAt = new Date(Date.now() + (tokens.expires_in - 60) * 1000).toISOString();
    }
  }
  settingsSave(env, settings);
}

/** Run the whole consent dance and store the resulting tokens. */
export async function connect(env: EnvRoot, connection: Connection, deps: SyncDeps): Promise<void> {
  const provider = providerFor(connection.provider);
  if (!provider) throw DddError.config(`${connection.provider} does not use OAuth`);
  const clientId = connection.config.clientId;
  if (!clientId) {
    throw DddError.config('This connection needs a client ID. Create an OAuth client of type "Desktop app" in the provider\'s console and paste it here.');
  }
  const clientSecret = connection.config.clientSecret ?? "";

  const server = createServer();
  await new Promise<void>((resolve) => server.listen(0, "127.0.0.1", resolve));
  const port = (server.address() as AddressInfo).port;
  const redirectUri = `http://127.0.0.1:${port}`;

  const verifier = randomToken();
  const challenge = pkceChallenge(verifier);
  const state = randomToken();

  let auth =
    `${provider.authUrl}?response_type=code&client_id=${encodeURIComponent(clientId)}` +
    `&redirect_uri=${encodeURIComponent(redirectUri)}&scope=${encodeURIComponent(provider.scopes)}` +
    `&state=${encodeURIComponent(state)}&code_challenge=${challenge}&code_challenge_method=S256`;
  for (const [key, value] of provider.extraAuthParams) auth += `&${key}=${encodeURIComponent(value)}`;

  const redirect = awaitRedirect(server, state);
  try {
    await deps.openUrl(auth);
  } catch (e) {
    server.close();
    throw DddError.config(`could not open a browser: ${e instanceof Error ? e.message : String(e)}`);
  }
  const code = await redirect;

  const tokens = await postForm(deps.fetch, provider.tokenUrl, [
    ["grant_type", "authorization_code"], ["code", code], ["redirect_uri", redirectUri],
    ["client_id", clientId], ["client_secret", clientSecret], ["code_verifier", verifier],
  ]);
  storeTokens(env, connection, tokens, deps.secrets);
}

/** The credential an adapter should use right now, refreshing first if the
 *  stored one is stale. Providers without OAuth fall straight through. */
export async function accessToken(env: EnvRoot, connection: Connection, deps: SyncDeps): Promise<string> {
  const stored = () => deps.secrets.get(env, connection.id) ?? "";
  const provider = providerFor(connection.provider);
  if (!provider) return stored();
  const expiresAt = connection.config.expiresAt;
  const fresh = expiresAt !== undefined && !Number.isNaN(Date.parse(expiresAt)) && Date.parse(expiresAt) > Date.now();
  if (fresh) return stored();
  const refresh = deps.secrets.get(env, `${connection.id}:refresh`);
  // No refresh token: never connected, or a provider that only issues
  // long-lived tokens (Slack). Use what we have.
  if (refresh === null) return stored();
  const tokens = await postForm(deps.fetch, provider.tokenUrl, [
    ["grant_type", "refresh_token"], ["refresh_token", refresh],
    ["client_id", connection.config.clientId ?? ""], ["client_secret", connection.config.clientSecret ?? ""],
  ]);
  storeTokens(env, connection, tokens, deps.secrets);
  return stored();
}
```

```ts
/**
 * The central scheduler. Port of `sync/scheduler.rs`.
 *
 * One loop for the whole app. Widgets declare what they need and read from
 * the cache; they never poll. Twelve widgets over one GitHub connection cost
 * exactly what one widget costs, which is the only way this survives contact
 * with API rate limits. This is the only writer of the item cache.
 */
import * as events from "../events";
import { globalLoad } from "../global";
import type { EnvRoot } from "../paths";
import { EnvRoot as EnvRootClass } from "../paths";
import { settingsLoad, settingsSave, type Connection, type ConnectionStatus } from "../settings";
import { forProvider, type FetchContext, type SyncDeps } from "./adapter";
import * as cache from "./cache";
import { accessToken } from "./oauth";

/** How often the loop wakes to see what is due. Not how often anything syncs —
 *  that is per connection, from settings. */
export const TICK_MS = 20_000;

/** Write sync outcome back onto the connection record. */
export function record(env: EnvRoot, connectionId: string, status: ConnectionStatus, error: string | null): void {
  const settings = settingsLoad(env);
  const connection = settings.connections.find((c) => c.id === connectionId);
  if (!connection) return;
  connection.status = status;
  connection.error = error;
  if (status === "connected") connection.lastSyncAt = new Date().toISOString();
  settingsSave(env, settings);
}

export class Scheduler {
  /** Connection id → when it last completed a fetch (ms since epoch). */
  private lastRun = new Map<string, number>();
  private timer: NodeJS.Timeout | null = null;
  private ticking = false;

  constructor(private readonly deps: SyncDeps) {}

  due(connectionId: string, everyMs: number): boolean {
    const last = this.lastRun.get(connectionId);
    // Never run: due immediately, so a freshly connected account fills the
    // board rather than making you wait out a refresh interval.
    if (last === undefined) return true;
    return Date.now() - last >= everyMs;
  }

  mark(connectionId: string): void {
    this.lastRun.set(connectionId, Date.now());
  }

  /** Force the next tick to treat this connection as due. */
  invalidate(connectionId: string): void {
    this.lastRun.delete(connectionId);
  }

  /** Sync one connection now and write the result into the cache. Returns how
   *  many items it contributed. Errors are recorded on the connection rather
   *  than thrown away, because "expired" is something the Connections page has
   *  to be able to show — and then rethrown, as the Rust's `Err(e)` is. */
  async syncConnection(env: EnvRoot, connection: Connection): Promise<number> {
    const adapter = forProvider(connection.provider);
    if (!adapter) return 0;

    // OAuth connections refresh here, transparently: an adapter never learns
    // whether its credential came from a consent flow or from a pasted token.
    const token = await accessToken(env, connection, this.deps).catch(() => "");
    // The mock adapter is the one source that needs no credential.
    if (token === "" && connection.provider !== "mock") {
      record(env, connection.id, "disconnected", "no token stored");
      return 0;
    }

    const cx: FetchContext = { environmentId: env.id, connection, token, fetch: this.deps.fetch };
    const payload = { environmentId: env.id, connectionId: connection.id };
    try {
      const items = await adapter.fetch(cx);
      cache.replaceForConnection(env, connection.id, items);
      record(env, connection.id, "connected", null);
      try {
        events.log(env, { kind: "sync.fetch", source: connection.provider, meta: { items: items.length } });
      } catch {
        /* a dropped stat never interrupts a sync */
      }
      this.deps.emit.emit("ddd://items-changed", payload);
      return items.length;
    } catch (e) {
      const message = e instanceof Error ? e.message : String(e);
      // 401/403 means the credential is the problem, which is a different
      // thing for the user to do about it than a network blip.
      const status: ConnectionStatus = message.includes("401") || message.includes("403") ? "expired" : "error";
      record(env, connection.id, status, message);
      this.deps.emit.emit("ddd://items-changed", payload);
      throw e;
    }
  }

  /** One pass over every environment's connections. */
  async tick(): Promise<void> {
    const global = globalLoad();
    // Refresh cadence is global: it is about how hard you want the machine
    // hitting the network, not about which environment you are looking at.
    const everyMs = Math.max(global.appearance.defaultRefreshSeconds, 30) * 1000;
    for (const environment of global.environments) {
      const env = EnvRootClass.open(environment.id);
      const settings = settingsLoad(env);
      for (const connection of settings.connections) {
        if (!this.due(connection.id, everyMs)) continue;
        // Marked before the fetch, not after: a slow or failing connection
        // must not be retried on every tick.
        this.mark(connection.id);
        await this.syncConnection(env, connection).catch(() => undefined);
      }
    }
  }

  /** The loop. Started once at app setup and left running. */
  start(): void {
    if (this.timer) return;
    const run = () => {
      if (this.ticking) return;
      this.ticking = true;
      this.tick()
        .catch((e) => console.error(`ddd: scheduler tick failed: ${e instanceof Error ? e.message : e}`))
        .finally(() => {
          this.ticking = false;
        });
    };
    run();
    this.timer = setInterval(run, TICK_MS);
  }

  stop(): void {
    if (this.timer) clearInterval(this.timer);
    this.timer = null;
  }
}
```
